4. Create SQL analysis file
In PyCharm, create:

analysis.sql

Put this inside:

USE sales_db;

-- 1. Display all sales
SELECT *
FROM sales;


-- 2. Calculate revenue for every order
SELECT
    order_id,
    order_date,
    product,
    category,
    quantity,
    price,
    quantity * price AS revenue
FROM sales;


-- 3. Calculate total revenue
SELECT
    SUM(quantity * price) AS total_revenue
FROM sales;


-- 4. Revenue by category
SELECT
    category,
    SUM(quantity * price) AS revenue
FROM sales
GROUP BY category
ORDER BY revenue DESC;


-- 5. Units sold by product
SELECT
    product,
    SUM(quantity) AS units_sold
FROM sales
GROUP BY product
ORDER BY units_sold DESC;


-- 6. Monthly revenue
SELECT
    DATE_FORMAT(order_date, '%Y-%m') AS month,
    SUM(quantity * price) AS revenue
FROM sales
GROUP BY DATE_FORMAT(order_date, '%Y-%m')
ORDER BY month;


-- 7. Best-selling product by revenue
SELECT
    product,
    SUM(quantity * price) AS revenue
FROM sales
GROUP BY product
ORDER BY revenue DESC
LIMIT 1;

6. Create database.py
This file connects Python to MySQL.

import mysql.connector


def get_connection():
    connection = mysql.connector.connect(
        host="localhost",
        user="root",
        password="YOUR_MYSQL_PASSWORD",
        database="sales_db"
    )

    return connection


if __name__ == "__main__":
    connection = get_connection()

    if connection.is_connected():
        print("Successfully connected to MySQL!")

    connection.close()

Change:

password="YOUR_MYSQL_PASSWORD"

to your actual MySQL password.

For example:

password="MyPassword123"

Run database.py.

You should see:

Successfully connected to MySQL!

7. Create main.py
Now we can use SQL + Python + Pandas together.

import mysql.connector
import pandas as pd
import matplotlib.pyplot as plt


# -----------------------------------------
# 1. Connect to MySQL
# -----------------------------------------

connection = mysql.connector.connect(
    host="localhost",
    user="root",
    password="YOUR_MYSQL_PASSWORD",
    database="sales_db"
)

print("Connected to MySQL successfully!")


# -----------------------------------------
# 2. SQL query
# -----------------------------------------

query = """
SELECT
    DATE_FORMAT(order_date, '%Y-%m') AS month,
    category,
    SUM(quantity * price) AS revenue,
    SUM(quantity) AS units_sold
FROM sales
GROUP BY
    DATE_FORMAT(order_date, '%Y-%m'),
    category
ORDER BY month, category
"""


# -----------------------------------------
# 3. Execute SQL and load into Pandas
# -----------------------------------------

df = pd.read_sql(query, connection)


# -----------------------------------------
# 4. Close database connection
# -----------------------------------------

connection.close()


# -----------------------------------------
# 5. Display data
# -----------------------------------------

print("\nSales Analysis")
print("=" * 50)

print(df)


# -----------------------------------------
# 6. Total Revenue
# -----------------------------------------

total_revenue = df["revenue"].sum()

print("\nTotal Revenue:")
print(total_revenue)


# -----------------------------------------
# 7. Revenue by Category
# -----------------------------------------

category_revenue = (
    df.groupby("category")["revenue"]
    .sum()
    .sort_values(ascending=False)
)

print("\nRevenue by Category:")
print(category_revenue)


# -----------------------------------------
# 8. Units Sold by Category
# -----------------------------------------

category_units = (
    df.groupby("category")["units_sold"]
    .sum()
    .sort_values(ascending=False)
)

print("\nUnits Sold by Category:")
print(category_units)


# -----------------------------------------
# 9. Monthly Revenue
# -----------------------------------------

df["month"] = pd.to_datetime(df["month"])

monthly_revenue = (
    df.groupby("month")["revenue"]
    .sum()
)

print("\nMonthly Revenue:")
print(monthly_revenue)


# -----------------------------------------
# 10. Create Chart
# -----------------------------------------

monthly_revenue.plot(
    kind="bar",
    color="steelblue"
)

plt.title("Monthly Revenue")
plt.xlabel("Month")
plt.ylabel("Revenue")

plt.xticks(rotation=45)

plt.tight_layout()

plt.show()
